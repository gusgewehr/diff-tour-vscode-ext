- stage {stage: export|write|check|show, state: active|done}. A cached run emits only `show done` then `done`. After a repair, `write active` is emitted again while `check` stays active; set states literally, never reset other stages.
- read {tool, path, tag: "" | "NOT_IN_DIFF" | "BASE"}
- repair {attempt: number}
- done {key, path, base, head, baseSha, headSha, files, cached}
- error {code: usage|git|not_signed_in|cancelled|invalid_tour|llm, message, problems?: string[]}, then exit 1

Contracts this plan CREATES (plans 02 and 03 import them exactly):
- src/cli/progress.ts
  - export type StageName = 'export' | 'write' | 'check' | 'show'; export const STAGES: StageName[] (in that order)
  - export type StageState = 'todo' | 'active' | 'done'
  - export interface ReadRow { time: string; tool: string; path: string; tag: string }
  - export interface DoneEvent { key: string; path: string; base: string; head: string; baseSha: string; headSha: string; files: number; cached: boolean }
  - export interface ErrorEvent { code: string; message: string; problems?: string[] }
  - export type CliEvent = { type: 'stage'; stage: StageName; state: 'active' | 'done' } | ({ type: 'read' } & Omit<ReadRow, 'time'>) | { type: 'repair'; attempt: number } | ({ type: 'done' } & DoneEvent) | ({ type: 'error' } & ErrorEvent)
  - export interface ProgressState { stages: Record<StageName, StageState>; reads: ReadRow[]; repairs: number; result?: DoneEvent; error?: ErrorEvent }
  - export function parseLine(line: string): CliEvent | undefined
  - export function initialState(): ProgressState
  - export function reduce(state: ProgressState, e: CliEvent, now?: Date): ProgressState
  - export function currentStage(s: ProgressState): string  (display name "Export" | "Write" | "Check" | "Show")
- src/cli/runner.ts
  - export interface SpawnSpec { command: string; args: string[]; cwd?: string }
  - export interface RunResult { exitCode: number | null; done?: DoneEvent; error?: ErrorEvent; cancelled: boolean; stderrTail: string[]; spawnError?: string }
  - export interface RunHandle { cancel(): void; done: Promise<RunResult> }
  - export function run(spec: SpawnSpec, onEvent: (e: CliEvent) => void, opts?: { log?: (line: string) => void; graceMs?: number }): RunHandle
  - export function cliLlm(llm: string): string
  - export function cliArgs(t: Pick<Target, 'kind' | 'base' | 'head' | 'baseSha' | 'headSha'>, s: Pick<Settings, 'llm' | 'language'>, repo: string): string[]
- src/cli/cache.ts
  - export interface TourMeta { key: string; label: string; base: string; head: string; baseSha: string; headSha: string; files: number; lang: string; llm: string; createdAt: string }
  - export async function saveTour(dir: string, done: DoneEvent, label: string, s: Pick<Settings, 'llm' | 'language'>): Promise<string>  (returns the cached tour path `<dir>/<key>.json`)
  - export async function findExisting(dir: string, t: Pick<Target, 'label' | 'baseSha' | 'headSha'>): Promise<TourMeta | undefined>
- src/cli/errors.ts
  - export type ErrorAction = 'Open Terminal' | 'Try again' | 'Show details' | 'Open Settings'
  - export function describeError(r: Pick<RunResult, 'error' | 'exitCode' | 'cancelled' | 'spawnError' | 'done'>): { text: string; actions: ErrorAction[] } | undefined
